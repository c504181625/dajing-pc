<script setup lang="ts">
import { ElMessage, ElMessageBox } from 'element-plus'
import { computed, reactive, ref } from 'vue'

import {
  deleteExpert,
  getCommunityExpertList,
  getCommunityOverview,
  saveCommunityExpert,
} from '@/api/modules/content'
import PageContainer from '@/components/PageContainer.vue'
import PermissionButton from '@/components-business/PermissionButton/index.vue'
import SearchForm from '@/components-business/SearchForm/index.vue'
import SectionCard from '@/components-business/SectionCard/index.vue'
import { ExpertServiceStatus } from '@/enum/content'
import { PERMISSION_CODE } from '@/enum/permission'
import type { CommunityOverviewData, ContentExpertForm, ExpertOnlineItem, ExpertQuery } from '@/types/content'

const loading = ref(false)
const total = ref(0)
const list = ref<ExpertOnlineItem[]>([])
const overview = ref<CommunityOverviewData | null>(null)
const detailVisible = ref(false)
const editVisible = ref(false)
const currentItem = ref<ExpertOnlineItem | null>(null)

const queryForm = reactive<ExpertQuery>({
  pageNum: 1,
  pageSize: 10,
  keyword: '',
  serviceStatus: '',
})

const form = reactive<(ContentExpertForm & { id?: string; specialtiesText?: string })>({
  id: '',
  name: '',
  title: '',
  organization: '',
  specialties: [],
  specialtiesText: '',
  serviceStatus: ExpertServiceStatus.Online,
  introduction: '',
  consultationPriceText: '',
  onlineTimeText: '',
})

const searchFields = computed(() => [
  { label: '关键�?, prop: 'keyword', placeholder: '专家姓名/机构/专长' },
  {
    label: '服务状�?,
    prop: 'serviceStatus',
    component: 'select' as const,
    placeholder: '服务状�?,
    options: [
      { label: '在线', value: ExpertServiceStatus.Online },
      { label: '忙碌', value: ExpertServiceStatus.Busy },
      { label: '离线', value: ExpertServiceStatus.Offline },
    ],
  },
])

const statCards = computed(() => [
  { title: '在线专家', value: overview.value?.expertTotal || 0 },
  { title: '资讯内容', value: overview.value?.articleTotal || 0 },
  { title: '问答内容', value: overview.value?.questionTotal || 0 },
])

async function loadOverview() {
  overview.value = await getCommunityOverview()
}

async function loadData() {
  loading.value = true
  try {
    const res = await getCommunityExpertList(queryForm)
    list.value = res.list
    total.value = res.total
  } finally {
    loading.value = false
  }
}

async function refreshAll() {
  await Promise.all([loadOverview(), loadData()])
}

function handleSearch() {
  queryForm.pageNum = 1
  loadData()
}

function handlePageChange(page: number) {
  queryForm.pageNum = page
  loadData()
}

function handleSizeChange(size: number) {
  queryForm.pageSize = size
  queryForm.pageNum = 1
  loadData()
}

function openCreate() {
  form.id = ''
  form.name = ''
  form.title = ''
  form.organization = ''
  form.specialties = []
  form.specialtiesText = ''
  form.serviceStatus = ExpertServiceStatus.Online
  form.introduction = ''
  form.consultationPriceText = ''
  form.onlineTimeText = ''
  editVisible.value = true
}

function openEdit(item: ExpertOnlineItem) {
  form.id = item.id
  form.name = item.name
  form.title = item.title
  form.organization = item.organization
  form.specialties = [...item.specialties]
  form.specialtiesText = item.specialties.join('�?)
  form.serviceStatus = item.serviceStatus
  form.introduction = item.introduction
  form.consultationPriceText = item.consultationPriceText || ''
  form.onlineTimeText = item.onlineTimeText || ''
  editVisible.value = true
}

function openDetail(item: ExpertOnlineItem) {
  currentItem.value = item
  detailVisible.value = true
}

async function handleSave() {
  const specialties = (form.specialtiesText || '')
    .split(/[,，、\s]+/)
    .map((item) => item.trim())
    .filter(Boolean)

  await saveCommunityExpert({
    id: form.id,
    name: form.name,
    title: form.title,
    organization: form.organization,
    specialties,
    serviceStatus: form.serviceStatus,
    introduction: form.introduction,
    consultationPriceText: form.consultationPriceText,
    onlineTimeText: form.onlineTimeText,
  })

  ElMessage.success(form.id ? '专家信息已更�? : '专家信息已新�?)
  editVisible.value = false
  await refreshAll()
}

async function handleDelete(item: ExpertOnlineItem) {
  try {
    await ElMessageBox.confirm(`确认删除专家�?{item.name}”吗？`, '删除确认', { type: 'warning' })
  } catch {
    return
  }
  await deleteExpert(item.id)
  ElMessage.success('专家信息已删�?)
  await refreshAll()
}

await refreshAll()
</script>

<template>
  <PageContainer title="专家在线" subtitle="维护专家展示、服务状态与咨询入口，恢复卡片化管理视图�?>
    <div class="hero-grid">
      <SectionCard v-for="item in statCards" :key="item.title">
        <div class="stat-card">
          <div class="stat-title">{{ item.title }}</div>
          <div class="stat-value">{{ item.value }}</div>
        </div>
      </SectionCard>
    </div>

    <SearchForm v-model="queryForm" :fields="searchFields" @search="handleSearch" @reset="loadData" />

    <SectionCard title="专家列表">
      <template #extra>
        <PermissionButton :permission="PERMISSION_CODE.contentManageView" @click="openCreate">
          新增专家
        </PermissionButton>
      </template>

      <div v-loading="loading" class="card-list">
        <article v-for="item in list" :key="item.id" class="content-card">
          <div class="card-head">
            <div class="card-tags">
              <el-tag
                :type="
                  item.serviceStatus === ExpertServiceStatus.Online
                    ? 'success'
                    : item.serviceStatus === ExpertServiceStatus.Busy
                      ? 'warning'
                      : 'info'
                "
                effect="light"
              >
                {{
                  item.serviceStatus === ExpertServiceStatus.Online
                    ? '在线'
                    : item.serviceStatus === ExpertServiceStatus.Busy
                      ? '忙碌'
                      : '离线'
                }}
              </el-tag>
            </div>
            <div class="card-actions">
              <el-button text type="primary" @click="openDetail(item)">查看详情</el-button>
              <PermissionButton :permission="PERMISSION_CODE.contentManageView" text @click="openEdit(item)">
                编辑
              </PermissionButton>
              <PermissionButton
                :permission="PERMISSION_CODE.contentManageView"
                text
                type="danger"
                @click="handleDelete(item)"
              >
                删除
              </PermissionButton>
            </div>
          </div>
          <h3 class="card-title">{{ item.name }} / {{ item.title }}</h3>
          <p class="card-summary">{{ item.organization }}</p>
          <div class="card-meta">
            <span>{{ item.specialties.join('�?) }}</span>
            <span>{{ item.consultationPriceText || '价格待定' }}</span>
            <span>{{ item.onlineTimeText || '在线时间待补�? }}</span>
          </div>
        </article>
      </div>

      <div v-if="total > 0" class="pagination-wrap">
        <el-pagination
          background
          layout="total, sizes, prev, pager, next, jumper"
          prev-text="上一�?
          next-text="下一�?
          :total="total"
          :current-page="queryForm.pageNum"
          :page-size="queryForm.pageSize"
          :page-sizes="[10, 20, 50]"
          @current-change="handlePageChange"
          @size-change="handleSizeChange"
        />
      </div>
    </SectionCard>

    <el-drawer v-model="detailVisible" title="专家详情" size="620px">
      <el-descriptions v-if="currentItem" :column="1" border>
        <el-descriptions-item label="姓名">{{ currentItem.name }}</el-descriptions-item>
        <el-descriptions-item label="职称">{{ currentItem.title }}</el-descriptions-item>
        <el-descriptions-item label="机构">{{ currentItem.organization }}</el-descriptions-item>
        <el-descriptions-item label="专长">{{ currentItem.specialties.join('�?) }}</el-descriptions-item>
        <el-descriptions-item label="介绍">{{ currentItem.introduction }}</el-descriptions-item>
      </el-descriptions>
    </el-drawer>

    <el-dialog v-model="editVisible" :title="form.id ? '编辑专家' : '新增专家'" width="620px">
      <el-form label-width="92px">
        <el-form-item label="姓名">
          <el-input v-model="form.name" />
        </el-form-item>
        <el-form-item label="职称">
          <el-input v-model="form.title" />
        </el-form-item>
        <el-form-item label="机构">
          <el-input v-model="form.organization" />
        </el-form-item>
        <el-form-item label="专长">
          <el-input v-model="form.specialtiesText" placeholder="多个专长请用顿号或逗号分隔" />
        </el-form-item>
        <el-form-item label="服务状�?>
          <el-select v-model="form.serviceStatus">
            <el-option label="在线" :value="ExpertServiceStatus.Online" />
            <el-option label="忙碌" :value="ExpertServiceStatus.Busy" />
            <el-option label="离线" :value="ExpertServiceStatus.Offline" />
          </el-select>
        </el-form-item>
        <el-form-item label="介绍">
          <el-input v-model="form.introduction" type="textarea" :rows="4" />
        </el-form-item>
      </el-form>
      <template #footer>
        <el-button @click="editVisible = false">取消</el-button>
        <el-button type="primary" @click="handleSave">保存</el-button>
      </template>
    </el-dialog>
  </PageContainer>
</template>
