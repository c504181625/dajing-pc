<script setup lang="ts">
import { ElMessage, ElMessageBox } from 'element-plus'
import { computed, reactive, ref } from 'vue'

import {
  deleteArticle,
  getCommunityArticleList,
  getCommunityOverview,
  getContentCategories,
  saveCommunityArticle,
} from '@/api/modules/content'
import PageContainer from '@/components/PageContainer.vue'
import PermissionButton from '@/components-business/PermissionButton/index.vue'
import SearchForm from '@/components-business/SearchForm/index.vue'
import SectionCard from '@/components-business/SectionCard/index.vue'
import { PERMISSION_CODE } from '@/enum/permission'
import { ContentBizType } from '@/enum/content'
import type { CommunityArticleItem, CommunityOverviewData, ContentArticleForm, ContentCategoryItem, ContentQuery } from '@/types/content'

const loading = ref(false)
const total = ref(0)
const list = ref<CommunityArticleItem[]>([])
const categories = ref<ContentCategoryItem[]>([])
const overview = ref<CommunityOverviewData | null>(null)
const detailVisible = ref(false)
const editVisible = ref(false)
const currentItem = ref<CommunityArticleItem | null>(null)

const queryForm = reactive<ContentQuery>({
  pageNum: 1,
  pageSize: 10,
  keyword: '',
  categoryCode: '',
  bizType: '',
})

const form = reactive<ContentArticleForm & { id?: string }>({
  id: '',
  bizType: ContentBizType.News,
  title: '',
  summary: '',
  categoryCode: '',
  featured: false,
})

const searchFields = computed(() => [
  { label: '关键�?, prop: 'keyword', placeholder: '搜索标题、摘要、作者或分类' },
  {
    label: '分类',
    prop: 'categoryCode',
    component: 'select' as const,
    placeholder: '选择分类',
    options: categories.value.map((item) => ({ label: item.name, value: item.code })),
  },
])

const statCards = computed(() => [
  { title: '已发布内�?, value: overview.value?.articleTotal || 0 },
  { title: '热门问答', value: overview.value?.questionTotal || 0 },
  { title: '在线专家', value: overview.value?.expertTotal || 0 },
])

async function loadOverview() {
  overview.value = await getCommunityOverview()
}

async function loadCategories() {
  categories.value = await getContentCategories()
}

async function loadData() {
  loading.value = true
  try {
    const res = await getCommunityArticleList(queryForm)
    list.value = res.list
    total.value = res.total
  } finally {
    loading.value = false
  }
}

async function refreshAll() {
  await Promise.all([loadOverview(), loadData()])
}

function handleSearch() {
  queryForm.pageNum = 1
  loadData()
}

function handlePageChange(page: number) {
  queryForm.pageNum = page
  loadData()
}

function handleSizeChange(size: number) {
  queryForm.pageSize = size
  queryForm.pageNum = 1
  loadData()
}

function openCreate() {
  form.id = ''
  form.bizType = ContentBizType.News
  form.title = ''
  form.summary = ''
  form.categoryCode = categories.value[0]?.code || ''
  form.featured = false
  editVisible.value = true
}

function openEdit(item: CommunityArticleItem) {
  form.id = item.id
  form.bizType = item.bizType
  form.title = item.title
  form.summary = item.summary
  form.categoryCode = item.categoryCode
  form.featured = !!item.featured
  editVisible.value = true
}

function openDetail(item: CommunityArticleItem) {
  currentItem.value = item
  detailVisible.value = true
}

async function handleSave() {
  await saveCommunityArticle(form)
  ElMessage.success(form.id ? '内容已更�? : '内容已新�?)
  editVisible.value = false
  await refreshAll()
}

async function handleDelete(item: CommunityArticleItem) {
  try {
    await ElMessageBox.confirm(`确认删除内容�?{item.title}”吗？`, '删除确认', { type: 'warning' })
  } catch {
    return
  }
  await deleteArticle(item.id)
  ElMessage.success('内容已删�?)
  await refreshAll()
}

await Promise.all([loadCategories(), refreshAll()])
</script>

<template>
  <PageContainer title="资讯中心" subtitle="覆盖平台公告、政策资讯与技术知识文章�?>
    <div class="hero-grid">
      <SectionCard v-for="item in statCards" :key="item.title">
        <div class="stat-card">
          <div class="stat-title">{{ item.title }}</div>
          <div class="stat-value">{{ item.value }}</div>
        </div>
      </SectionCard>
    </div>

    <SearchForm v-model="queryForm" :fields="searchFields" @search="handleSearch" @reset="loadData" />

    <SectionCard title="内容列表">
      <template #extra>
        <PermissionButton :permission="PERMISSION_CODE.contentManageView" @click="openCreate">
          新增内容
        </PermissionButton>
      </template>

      <div v-loading="loading" class="card-list">
        <article v-for="item in list" :key="item.id" class="content-card">
          <div class="card-head">
            <div class="card-tags">
              <el-tag type="primary" effect="light">{{ item.categoryName }}</el-tag>
              <el-tag v-if="item.featured" type="warning" effect="light">精�?/el-tag>
            </div>
            <div class="card-actions">
              <el-button text type="primary" @click="openDetail(item)">查看详情</el-button>
              <PermissionButton :permission="PERMISSION_CODE.contentManageView" text @click="openEdit(item)">
                编辑
              </PermissionButton>
              <PermissionButton
                :permission="PERMISSION_CODE.contentManageView"
                text
                type="danger"
                @click="handleDelete(item)"
              >
                删除
              </PermissionButton>
            </div>
          </div>
          <h3 class="card-title">{{ item.title }}</h3>
          <p class="card-summary">{{ item.summary }}</p>
          <div class="card-meta">
            <span>{{ item.author.name }}</span>
            <span>{{ item.publishTime }}</span>
            <span>{{ item.viewCount }} 浏览</span>
          </div>
        </article>
      </div>

      <div v-if="total > 0" class="pagination-wrap">
        <el-pagination
          background
          layout="total, sizes, prev, pager, next, jumper"
          prev-text="上一�?
          next-text="下一�?
          :total="total"
          :current-page="queryForm.pageNum"
          :page-size="queryForm.pageSize"
          :page-sizes="[10, 20, 50]"
          @current-change="handlePageChange"
          @size-change="handleSizeChange"
        />
      </div>
    </SectionCard>

    <el-drawer v-model="detailVisible" title="内容详情" size="620px">
      <el-descriptions v-if="currentItem" :column="1" border>
        <el-descriptions-item label="标题">{{ currentItem.title }}</el-descriptions-item>
        <el-descriptions-item label="摘要">{{ currentItem.summary }}</el-descriptions-item>
        <el-descriptions-item label="分类">{{ currentItem.categoryName }}</el-descriptions-item>
        <el-descriptions-item label="作�?>{{ currentItem.author.name }}</el-descriptions-item>
        <el-descriptions-item label="发布时间">{{ currentItem.publishTime }}</el-descriptions-item>
      </el-descriptions>
    </el-drawer>

    <el-dialog v-model="editVisible" :title="form.id ? '编辑内容' : '新增内容'" width="560px">
      <el-form label-width="92px">
        <el-form-item label="类型">
          <el-select v-model="form.bizType">
            <el-option label="资讯" :value="ContentBizType.News" />
            <el-option label="知识" :value="ContentBizType.Knowledge" />
          </el-select>
        </el-form-item>
        <el-form-item label="标题">
          <el-input v-model="form.title" />
        </el-form-item>
        <el-form-item label="摘要">
          <el-input v-model="form.summary" type="textarea" :rows="4" />
        </el-form-item>
        <el-form-item label="分类">
          <el-select v-model="form.categoryCode">
            <el-option
              v-for="item in categories"
              :key="item.code"
              :label="item.name"
              :value="item.code"
            />
          </el-select>
        </el-form-item>
      </el-form>
      <template #footer>
        <el-button @click="editVisible = false">取消</el-button>
        <el-button type="primary" @click="handleSave">保存</el-button>
      </template>
    </el-dialog>
  </PageContainer>
</template>
